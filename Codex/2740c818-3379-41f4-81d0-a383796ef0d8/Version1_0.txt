import java.util.Arrays;

public class MergeSortedLists {

    public static void main(String[] args) {
        // Test case 1: Merge two sorted lists
        int[] list1 = {1, 3, 7, 9, 13, 40};
        int[] list2 = {2, 4, 6, 14, 19, 23};
        int[] mergedList = mergeLists(list1, list2);
        System.out.println("Merged List: " + Arrays.toString(mergedList));

        // Test case 2: Merge two empty lists
        int[] list3 = {};
        int[] list4 = {};
        int[] mergedList2 = mergeLists(list3, list4);
        System.out.println("Merged List: " + Arrays.toString(mergedList2));

        // Test case 3: Merge a non-empty list with an empty list
        int[] list5 = {1, 3, 7, 9, 13, 40};
        int[] list6 = {};
        int[] mergedList3 = mergeLists(list5, list6);
        System.out.println("Merged List: " + Arrays.toString(mergedList3));

        // Test case 4: Remove all occurrences of a specified value from an array
        int[] arr = {1, 4, 6, 7, 6, 2};
        int value = 6;
        int[] updatedArray = removeValue(arr, value);
        System.out.println("Updated array: " + Arrays.toString(updatedArray));
        System.out.println("The length of the new array is: " + updatedArray.length);

        // Test case 5: Remove all occurrences of a specified value from an array with no occurrences
        int[] arr2 = {1, 4, 7, 2};
        int value2 = 6;
        int[] updatedArray2 = removeValue(arr2, value2);
        System.out.println("Updated array: " + Arrays.toString(updatedArray2));
        System.out.println("The length of the new array is: " + updatedArray2.length);

        // Test case 6: Remove all occurrences of a specified value from an empty array
        int[] arr3 = {};
        int value3 = 6;
        int[] updatedArray3 = removeValue(arr3, value3);
        System.out.println("Updated array: " + Arrays.toString(updatedArray3));
        System.out.println("The length of the new array is: " + updatedArray3.length);
    }

    // Function to merge two sorted lists
    public static int[] mergeLists(int[] list1, int[] list2) {
        int[] mergedList = new int[list1.length + list2.length];
        int i = 0, j = 0, k = 0;

        while (i < list1.length && j < list2.length) {
            if (list1[i] < list2[j]) {
                mergedList[k] = list1[i];
                i++;
            } else {
                mergedList[k] = list2[j];
                j++;
            }
            k++;
        }

        while (i < list1.length) {
            mergedList[k] = list1[i];
            i++;
            k++;
        }

        while (j < list2.length) {
            mergedList[k] = list2[j];
            j++;
            k++;
        }

        return mergedList;
    }

    // Function to remove all occurrences of a specified value from an array
    public static int[] removeValue(int[] arr, int value) {
        int[] updatedArray = new int[arr.length];
        int k = 0;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] != value) {
                updatedArray[k] = arr[i];
                k++;
            }
        }

        return Arrays.copyOf(updatedArray, k);
    }
}

    public static int findLargestPrimeFactor(int number, PrimeNumberChecker primeNumberChecker) {
        int largestPrimeFactor = -1;

        // Iterate from 2 to the square root of the number
        for (int i = 2; i <= Math.sqrt(number); i++) {
            // Check if the number is divisible by the current factor
            if (number % i == 0) {
                // Check if the factor is prime
                if (primeNumberChecker.isPrime(i)) {
                    // Update the largest prime factor if the current factor is larger
                    largestPrimeFactor = i;
                }
            }
        }

public class Exponentiation {

    public static int exponentiation(int base, int exponent) {
        if (exponent == 0) {
            return 1;
        } else if (exponent < 0) {
            return 1 / exponentiation(base, -exponent);
        } else {
            return base * exponentiation(base, exponent - 1);
        }
    }

    public static void main(String[] args) {
        System.out.println(exponentiation(2, 4)); // 16
        System.out.println(exponentiation(3, 3)); // 27
        System.out.println(exponentiation(5, 2)); // 25
        System.out.println(exponentiation(10, 0)); // 1
        System.out.println(exponentiation(2, -3)); // 0.125
    }
}

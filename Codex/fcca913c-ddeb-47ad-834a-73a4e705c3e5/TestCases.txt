**Java Program:**

```java
import java.util.Scanner;

public class LongestSubstringAtEnds {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Enter a string: ");
        String string = scanner.nextLine();

        // Find the longest substring that appears at both ends
        String longestSubstring = "";
        for (int i = 0; i < string.length() / 2; i++) {
            String substring = string.substring(i, string.length() - i);
            if (string.startsWith(substring) && string.endsWith(substring)) {
                if (substring.length() > longestSubstring.length()) {
                    longestSubstring = substring;
                }
            }
        }

        // Print the longest substring
        System.out.println("The longest substring that appears at both ends: " + longestSubstring);
    }
}
```

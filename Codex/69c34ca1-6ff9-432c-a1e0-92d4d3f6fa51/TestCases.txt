**Java Program:**

```java
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class StringContainsAnotherString {

    public static boolean containsAllLetters(String str1, String str2) {
        // Check if str2 is empty or longer than str1
        if (str2.length() == 0 || str2.length() > str1.length()) {
            return false;
        }

        // Convert both strings to lowercase for case-insensitive comparison
        str1 = str1.toLowerCase();
        str2 = str2.toLowerCase();

        // Create a map to store the frequency of characters in str1
        Map<Character, Integer> charFrequency = new HashMap<>();
        for (char c : str1.toCharArray()) {
            charFrequency.put(c, charFrequency.getOrDefault(c, 0) + 1);
        }

        // Iterate over str2 and check if each character is present in charFrequency
        for (char c : str2.toCharArray()) {
            if (!charFrequency.containsKey(c) || charFrequency.get(c) <= 0) {
                return false;
            }

            charFrequency.put(c, charFrequency.get(c) - 1);
        }

        // All letters in str2 are present in str1
        return true;
    }

    public static void main(String[] args) {
        // Test cases
        String[] str1 = {"Hello", "world", "programming", ""};
        String[] str2 = {"ell", "worl", "pro", "ab"};
        for (int i = 0; i < str1.length; i++) {
            System.out.println("Input Strings: " + str1[i] + ", " + str2[i]);
            System.out.println("Output: " + containsAllLetters(str1[i], str2[i]));
            System.out.println();
        }
    }
}
```

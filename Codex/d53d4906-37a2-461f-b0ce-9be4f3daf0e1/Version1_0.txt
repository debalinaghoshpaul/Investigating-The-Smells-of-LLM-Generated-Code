```java
import java.util.ArrayList;
import java.util.List;

public class InterleavingStrings {

    public static void main(String[] args) {
        String s1 = "AB";
        String s2 = "CD";
        List<String> interleavings = new ArrayList<>();
        interleave(s1, s2, "", interleavings);
        System.out.println(interleavings);
    }

    public static void interleave(String s1, String s2, String interleaving, List<String> interleavings) {
        if (s1.isEmpty() && s2.isEmpty()) {
            interleavings.add(interleaving);
            return;
        }

        if (!s1.isEmpty()) {
            interleave(s1.substring(1), s2, interleaving + s1.charAt(0), interleavings);
        }

        if (!s2.isEmpty()) {
            interleave(s1, s2.substring(1), interleaving + s2.charAt(0), interleavings);
        }
    }
}
```

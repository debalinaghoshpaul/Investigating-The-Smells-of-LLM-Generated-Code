```java
import java.util.*;

public class FindRowColPosition {

    public static void main(String[] args) {
        int[][] matrix = {
                {1, 3, 5, 7},
                {10, 11, 16, 20},
                {23, 30, 34, 50}
        };

        int target = 3;
        int[] result = findRowColPosition(matrix, target);

        if (result[0] == -1 && result[1] == -1) {
            System.out.println("Target not found");
        } else {
            System.out.println("Target found at row " + result[0] + ", column " + result[1]);
        }
    }

    public static int[] findRowColPosition(int[][] matrix, int target) {
        int m = matrix.length;
        int n = matrix[0].length;

        int row = 0;
        int col = n - 1;

        while (row < m && col >= 0) {
            if (matrix[row][col] == target) {
                return new int[]{row, col};
            } else if (matrix[row][col] < target) {
                row++;
            } else {
                col--;
            }
        }

        return new int[]{-1, -1};
    }
}
```

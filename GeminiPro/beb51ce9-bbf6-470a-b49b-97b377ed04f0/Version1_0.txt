import java.util.ArrayList;
import java.util.List;

public class ExtractPortionOfArrayList {

    public static void main(String[] args) {
        // Create an ArrayList
        ArrayList<Integer> numbers = new ArrayList<>();
        numbers.add(1);
        numbers.add(2);
        numbers.add(3);
        numbers.add(4);
        numbers.add(5);
        numbers.add(6);
        numbers.add(7);
        numbers.add(8);
        numbers.add(9);
        numbers.add(10);

        // Extract a portion of the ArrayList using subList() method
        List<Integer> portion = numbers.subList(2, 5);

        // Print the extracted portion
        System.out.println(portion);
    }
}

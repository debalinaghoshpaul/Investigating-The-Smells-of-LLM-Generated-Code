import java.util.Map;
import java.util.TreeMap;

public class LeastKeyGreaterthanGivenKey {

    public static void main(String[] args) {
        Map<String, Integer> map = new TreeMap<>();
        map.put("A", 1);
        map.put("B", 2);
        map.put("C", 3);
        map.put("D", 4);
        map.put("E", 5);

        String key = "C";

        // Get the least key strictly greater than the given key
        String leastKeyGreaterThanGivenKey = map.ceilingKey(key);

        // Print the least key strictly greater than the given key
        System.out.println(leastKeyGreaterThanGivenKey); // Output: D
    }
}

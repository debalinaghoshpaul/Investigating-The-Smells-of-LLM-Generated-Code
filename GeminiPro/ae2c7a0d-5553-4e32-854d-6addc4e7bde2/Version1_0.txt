import java.util.Stack;

public class StackCommonElements {

    public static void main(String[] args) {
        // Create two stacks
        Stack<Integer> stack1 = new Stack<>();
        Stack<Integer> stack2 = new Stack<>();

        // Push elements into stack1
        stack1.push(1);
        stack1.push(2);
        stack1.push(3);
        stack1.push(4);
        stack1.push(5);

        // Push elements into stack2
        stack2.push(3);
        stack2.push(4);
        stack2.push(5);
        stack2.push(6);
        stack2.push(7);

        // Find common elements between two stacks
        Stack<Integer> commonElements = findCommonElements(stack1, stack2);

        // Print common elements
        System.out.println("Common elements: " + commonElements);
    }

    public static Stack<Integer> findCommonElements(Stack<Integer> stack1, Stack<Integer> stack2) {
        Stack<Integer> commonElements = new Stack<>();

        // Iterate over stack1
        while (!stack1.isEmpty()) {
            int element1 = stack1.pop();

            // Iterate over stack2
            while (!stack2.isEmpty()) {
                int element2 = stack2.pop();

                // If the elements are equal, add them to the common elements stack
                if (element1 == element2) {
                    commonElements.push(element1);
                }
            }

            // Push stack2 back to its original order
            while (!stack2.isEmpty()) {
                stack2.push(stack2.pop());
            }
        }

        return commonElements;
    }
}

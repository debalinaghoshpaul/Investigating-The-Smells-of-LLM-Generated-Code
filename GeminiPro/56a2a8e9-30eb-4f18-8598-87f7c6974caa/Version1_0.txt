```java
import java.util.HashMap;
import java.util.Map;

public class IsomorphicStrings {

    public static boolean isIsomorphic(String str1, String str2) {
        if (str1 == null || str2 == null || str1.length() != str2.length()) {
            return false;
        }

        Map<Character, Character> map = new HashMap<>();

        for (int i = 0; i < str1.length(); i++) {
            char ch1 = str1.charAt(i);
            char ch2 = str2.charAt(i);

            if (map.containsKey(ch1)) {
                if (map.get(ch1) != ch2) {
                    return false;
                }
            } else {
                if (map.containsValue(ch2)) {
                    return false;
                }

                map.put(ch1, ch2);
            }
        }

        return true;
    }

    public static void main(String[] args) {
        String str1 = "egg";
        String str2 = "add";

        System.out.println(isIsomorphic(str1, str2)); // true

        str1 = "foo";
        str2 = "bar";

        System.out.println(isIsomorphic(str1, str2)); // false

        str1 = "paper";
        str2 = "title";

        System.out.println(isIsomorphic(str1, str2)); // true

        str1 = null;
        str2 = "title";

        System.out.println(isIsomorphic(str1, str2)); // false

        str1 = "egg";
        str2 = null;

        System.out.println(isIsomorphic(str1, str2)); // false

        str1 = "";
        str2 = "";

        System.out.println(isIsomorphic(str1, str2)); // true

        str1 = "ab";
        str2 = "ca";

        System.out.println(isIsomorphic(str1, str2)); // true

        str1 = "ab";
        str2 = "aa";

        System.out.println(isIsomorphic(str1, str2)); // false

        str1 = "ab";
        str2 = "abc";

        System.out.println(isIsomorphic(str1, str2)); // false
    }
}
```

**Normal Cases:**

* Input: str1 = "egg", str2 = "add"
* Expected Output: true

* Input: str1 = "foo", str2 = "bar"
* Expected Output: false

* Input: str1 = "paper", str2 = "title"
* Expected Output: true

**Corner Cases:**

* Input: str1 = null, str2 = "title"
* Expected Output: false

* Input: str1 = "egg", str2 = null
* Expected Output: false

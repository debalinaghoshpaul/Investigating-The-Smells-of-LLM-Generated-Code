**Java Program:**

```java
import java.util.Scanner;

public class CountLifeSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Get the input string
        System.out.println("Enter a string: ");
        String inputString = scanner.nextLine();

        // Count the number of occurrences of "life" and "li?e"
        int lifeCount = 0;
        int lifeAnyCount = 0;

        for (int i = 0; i < inputString.length(); i++) {
            String substring = inputString.substring(i, Math.min(i + 3, inputString.length()));
            if (substring.equals("life")) {
                lifeCount++;
            } else if (substring.matches("li.*e")) {
                lifeAnyCount++;
            }
        }

        // Print the results
        System.out.println("The number of occurrences of \"life\" is: " + lifeCount);
        System.out.println("The number of occurrences of \"li?e\" is: " + lifeAnyCount);
    }
}
```

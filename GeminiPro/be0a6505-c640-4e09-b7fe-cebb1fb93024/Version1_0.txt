```java
import java.util.Arrays;
import java.util.Random;

public class ShuffleArray {

    public static void main(String[] args) {
        int[] nums = {1, 2, 3, 4, 5, 6};
        System.out.println("Original Array: " + Arrays.toString(nums));

        // Shuffle the array
        shuffleArray(nums);

        // Print the shuffled array
        System.out.println("Shuffled Array: " + Arrays.toString(nums));
    }

    public static void shuffleArray(int[] nums) {
        // Create a random object
        Random rand = new Random();

        // Iterate over the array
        for (int i = 0; i < nums.length; i++) {
            // Generate a random index
            int randomIndex = rand.nextInt(nums.length);

            // Swap the elements at the current index and the random index
            int temp = nums[i];
            nums[i] = nums[randomIndex];
            nums[randomIndex] = temp;
        }
    }

    // Test cases
    public static void testCases() {
        // Empty array
        int[] emptyArray = {};
        shuffleArray(emptyArray);
        System.out.println("Shuffled Empty Array: " + Arrays.toString(emptyArray));

        // Array with one element
        int[] singleElementArray = {1};
        shuffleArray(singleElementArray);
        System.out.println("Shuffled Single Element Array: " + Arrays.toString(singleElementArray));

        // Array with duplicate elements
        int[] duplicateArray = {1, 2, 3, 4, 5, 1, 2, 3};
        shuffleArray(duplicateArray);
        System.out.println("Shuffled Array with Duplicates: " + Arrays.toString(duplicateArray));
    }
}
```

**Test cases including corner cases:**

* **Empty array:** An empty array should not be modified.
* **Array with one element:** An array with one element should not be modified.
* **Array with duplicate elements:** The shuffled array should still have the same number of duplicate elements.

**Java Program:**

```java
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;

public class DistinctIdentifier {

    public static String getDistinctIdentifier(String input) {
        try {
            MessageDigest md5 = MessageDigest.getInstance("MD5");
            byte[] digest = md5.digest(StandardCharsets.UTF_8.encode(input).array());
            return new String(digest).toUpperCase();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }

    public static void main(String[] args) {
        String input = "Hello World";
        System.out.println("Distinct identifier for \"" + input + "\": " + getDistinctIdentifier(input));
    }
}
```

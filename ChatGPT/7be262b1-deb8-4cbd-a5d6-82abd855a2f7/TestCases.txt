```java
import java.util.Scanner;

public class RemoveWordFromString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Get the input string and the word to be removed
        System.out.println("Enter the input string:");
        String inputString = scanner.nextLine();
        System.out.println("Enter the word to be removed:");
        String wordToRemove = scanner.nextLine();

        // Remove the specified word from the input string
        String updatedString = removeWord(inputString, wordToRemove);

        // Print the updated string
        System.out.println("Updated string:");
        System.out.println(updatedString);
    }

    public static String removeWord(String inputString, String wordToRemove) {
        // Check if the input string is empty or null
        if (inputString == null || inputString.isEmpty()) {
            return "";
        }

        // Check if the word to be removed is empty or null
        if (wordToRemove == null || wordToRemove.isEmpty()) {
            return inputString;
        }

        // Split the input string into an array of words
        String[] words = inputString.split(" ");

        // Create a new array to store the updated words
        String[] updatedWords = new String[words.length];

        // Copy the words from the input string to the updated array, skipping the word to be removed
        int index = 0;
        for (String word : words) {
            if (!word.equals(wordToRemove)) {
                updatedWords[index++] = word;
            }
        }

        // Join the updated words into a single string
        String updatedString = String.join(" ", updatedWords);

        // Return the updated string
        return updatedString;
    }
}
```

**Test Cases:**

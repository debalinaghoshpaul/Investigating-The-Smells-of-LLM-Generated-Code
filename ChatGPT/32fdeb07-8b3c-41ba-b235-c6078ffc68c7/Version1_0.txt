```java
import java.util.HashMap;
import java.util.Map;

public class GetPortionOfMap {

    public static void main(String[] args) {
        // create a map
        Map<String, Integer> map = new HashMap<>();
        map.put("A", 1);
        map.put("B", 2);
        map.put("C", 3);
        map.put("D", 4);
        map.put("E", 5);

        // get the portion of the map whose keys range from "B" to "D"
        Map<String, Integer> portion = map.subMap("B", "D");

        // print the portion of the map
        System.out.println(portion);
    }
}
```

// Java program to find the majority element in an array

import java.util.HashMap;

public class MajorityElement {

    public static int findMajorityElement(int[] numbers) {
        HashMap<Integer, Integer> map = new HashMap<>();

        for (int num : numbers) {
            if (map.containsKey(num)) {
                map.put(num, map.get(num) + 1);
            } else {
                map.put(num, 1);
            }
        }

        int majorityElement = -1;
        int majorityCount = numbers.length / 2;

        for (int key : map.keySet()) {
            if (map.get(key) > majorityCount) {
                majorityElement = key;
                break;
            }
        }

        return majorityElement;
    }

    public static void main(String[] args) {
        int[] numbers = {1, 2, 2, 2, 3, 2, 4, 2, 2};
        int majorityElement = findMajorityElement(numbers);

        if (majorityElement != -1) {
            System.out.println("Majority Element: " + majorityElement);
        } else {
            System.out.println("No majority element found.");
        }
    }
}

**Java Program:**

```java
import java.util.Scanner;

public class GetCharacterAtIndex {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        // Get the string from the user
        System.out.println("Enter a string: ");
        String str = input.nextLine();

        // Get the index from the user
        System.out.println("Enter the index of the character (0-based): ");
        int index = input.nextInt();

        // Check if the index is valid
        if (index < 0 || index >= str.length()) {
            System.out.println("Invalid index");
        } else {
            // Get the character at the given index
            char ch = str.charAt(index);

            // Print the character (Unicode code point)
            System.out.println("The character at index " + index + " is: " + ch + " (Unicode code point: " + (int) ch + ")");
        }
    }
}
```

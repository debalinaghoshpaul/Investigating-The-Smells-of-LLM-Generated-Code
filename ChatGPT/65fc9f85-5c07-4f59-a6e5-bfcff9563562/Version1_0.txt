import java.util.TreeSet;

public class FindNumbersLessThan7 {

    public static void main(String[] args) {
        // Create a tree set
        TreeSet<Integer> numbers = new TreeSet<>();

        // Add some numbers to the tree set
        numbers.add(1);
        numbers.add(2);
        numbers.add(3);
        numbers.add(4);
        numbers.add(5);
        numbers.add(6);
        numbers.add(7);

        // Find numbers less than 7 in the tree set
        TreeSet<Integer> numbersLessThan7 = numbers.headSet(7);

        // Print the numbers less than 7
        System.out.println(numbersLessThan7);
    }
}

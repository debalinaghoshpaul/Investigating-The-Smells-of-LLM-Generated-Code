    public static int[] createArrayBeforeTen(int[] arr) {
        // Find the index of the last element with value 10
        int lastTenIndex = -1;
        for (int i = arr.length - 1; i >= 0; i--) {
            if (arr[i] == 10) {
                lastTenIndex = i;
                break;
            }
        }

        // If there is no element with value 10, return the original array
        if (lastTenIndex == -1) {
            return arr;
        }

        // Create a new array to store the elements before the last 10
        int[] newArr = new int[lastTenIndex];
        for (int i = 0; i < lastTenIndex; i++) {
            newArr[i] = arr[i];
        }

        return newArr;
    }

    public static boolean checkSameNumbersAtStartAndEnd(int[] arr, int[] start, int[] end) {
        // Check if the start and end arrays have the same length
        if (start.length != end.length) {
            return false;
        }

        // Check if the start and end arrays are empty
        if (start.length == 0) {
            return true;
        }

        // Check if the first elements of the start and end arrays are the same
        if (start[0] != end[0]) {
            return false;
        }

        // Check if the last elements of the start and end arrays are the same
        if (start[start.length - 1] != end[end.length - 1]) {
            return false;
        }

        // Iterate over the remaining elements of the start and end arrays to check if they are the same
        for (int i = 1; i < start.length - 1; i++) {
            if (start[i] != end[i]) {
                return false;
            }
        }

        // If all the elements of the start and end arrays are the same, return true
        return true;
    }

```java
import java.util.ArrayList;

public class IncreaseArrayListSize {

    public static void main(String[] args) {
        // Create an array list with an initial capacity of 5
        ArrayList<Integer> numbers = new ArrayList<>(5);

        // Add some elements to the array list
        numbers.add(1);
        numbers.add(2);
        numbers.add(3);
        numbers.add(4);
        numbers.add(5);

        // Increase the capacity of the array list to 10
        numbers.ensureCapacity(10);

        // Add some more elements to the array list
        numbers.add(6);
        numbers.add(7);
        numbers.add(8);
        numbers.add(9);
        numbers.add(10);

        // Print the contents of the array list
        System.out.println(numbers);
    }
}
```

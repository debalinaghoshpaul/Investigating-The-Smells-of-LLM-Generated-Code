```java
import java.util.*;

public class DistinctCharacters {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Get the input string from the user
        System.out.println("Enter a string: ");
        String str = scanner.nextLine();

        // Create a set to store the characters of the string
        Set<Character> characters = new HashSet<>();

        // Iterate over the string and add each character to the set
        for (char c : str.toCharArray()) {
            characters.add(c);
        }

        // Check if the size of the set is equal to the length of the string
        if (characters.size() == str.length()) {
            System.out.println("String has all unique characters: true");
        } else {
            System.out.println("String has all unique characters: false");
        }
    }
}
```

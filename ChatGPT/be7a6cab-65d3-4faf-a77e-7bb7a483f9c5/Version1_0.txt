```java
import java.util.TreeSet;

public class GetElementLowerThan {

    public static void main(String[] args) {
        // create a tree set
        TreeSet<Integer> treeSet = new TreeSet<>();

        // add some values to the tree set
        treeSet.add(1);
        treeSet.add(2);
        treeSet.add(3);
        treeSet.add(4);
        treeSet.add(5);

        // get an element in the tree set that has a lower value than the given element
        Integer lowerElement = treeSet.lower(3);

        // print the lower element
        System.out.println(lowerElement); // 2
    }
}
```

public class CountTriples {

    public static void main(String[] args) {

        // Get the input string from the user.
        System.out.println("Enter a string: ");
        String str = scanner.nextLine();

        // Initialize the count of triples to 0.
        int count = 0;

        // Iterate over the string, starting from the second character.
        for (int i = 1; i < str.length(); i++) {
            // Check if the current character is the same as the previous two characters.
            if (str.charAt(i) == str.charAt(i - 1) && str.charAt(i) == str.charAt(i - 2)) {
                // If so, increment the count of triples.
                count++;
            }
        }

        // Print the count of triples.
        System.out.println("The number of triples in the string is: " + count);
    }
}

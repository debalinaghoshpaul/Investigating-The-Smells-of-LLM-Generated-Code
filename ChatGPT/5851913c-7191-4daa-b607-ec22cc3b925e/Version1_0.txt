import java.util.Arrays;

public class ArrayComparison {

    public static void main(String[] args) {

      // Test case 1: Two arrays with the same length and some elements that differ by 1 or less
      int[] arr1 = {1, 2, 3, 4, 5};
      int[] arr2 = {2, 3, 4, 5, 6};
      int result1 = countDifferences(arr1, arr2);
      System.out.println("Number of elements that differ by 1 or less in the first test case: " + result1); // Expected: 4

      // Test case 2: Two arrays with the same length and no elements that differ by 1 or less
      int[] arr3 = {1, 3, 5, 7, 9};
      int[] arr4 = {2, 4, 6, 8, 10};
      int result2 = countDifferences(arr3, arr4);
      System.out.println("Number of elements that differ by 1 or less in the second test case: " + result2); // Expected: 0

      // Test case 3: Two arrays with different lengths
      int[] arr5 = {1, 2, 3};
      int[] arr6 = {2, 3, 4, 5};
      int result3 = countDifferences(arr5, arr6);
      System.out.println("Number of elements that differ by 1 or less in the third test case: " + result3); // Expected: -1

      // Test case 4: Two arrays with null values
      int[] arr7 = null;
      int[] arr8 = {1, 2, 3};
      int result4 = countDifferences(arr7, arr8);
      System.out.println("Number of elements that differ by 1 or less in the fourth test case: " + result4); // Expected: -1

      // Test case 5: Corner case with two empty arrays
      int[] arr9 = {};
      int[] arr10 = {};
      int result5 = countDifferences(arr9, arr10);
      System.out.println("Number of elements that differ by 1 or less in the fifth test case: " + result5); // Expected: 0

      // Test case 6: Corner case with two arrays containing only one element that differs by 1
      int[] arr11 = {1};
      int[] arr12 = {2};
      int result6 = countDifferences(arr11, arr12);
      System.out.println("Number of elements that differ by 1 or less in the sixth test case: " + result6); // Expected: 1

    }

    public static int countDifferences(int[] arr1, int[] arr2) {
        if (arr1 == null || arr2 == null || arr1.length != arr2.length) {
            return -1;
        }

        int count = 0;
        for (int i = 0; i < arr1.length; i++) {
            if (Math.abs(arr1[i] - arr2[i]) <= 1) {
                count++;
            }
        }

        return count;
    }
}

import java.util.HashMap;
import java.util.Map;
import java.util.TreeMap;

public class GetPortionOfMapGreaterThanOrEqualTo {

    public static void main(String[] args) {
        // create a map
        Map<String, Integer> map = new HashMap<>();
        map.put("apple", 10);
        map.put("banana", 20);
        map.put("cherry", 30);
        map.put("durian", 40);
        map.put("elderberry", 50);

        // create a sorted map from the original map
        TreeMap<String, Integer> sortedMap = new TreeMap<>(map);

        // get a portion of the map whose keys are greater than or equal to a given key
        Map<String, Integer> portionMap = sortedMap.tailMap("cherry");

        // print the portion map
        System.out.println(portionMap);
    }
}

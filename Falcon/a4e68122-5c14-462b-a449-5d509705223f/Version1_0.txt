import java.util.Scanner;

public class LongestZeroSequence {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int maxLength = 0;
        int maxIndex = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i]!= 0) {
                if (arr[i] > maxLength) {
                    maxLength = arr[i];
                    maxIndex = i;
                }
            }
        }
        System.out.println(maxLength);
        System.out.println(maxIndex);
    }
}

import java.util.Date;

public class DateDiff {
    public static void main(String[] args) {
        Date date1 = new Date();
        Date date2 = new Date();
        int year1 = date1.getYear();
        int month1 = date1.getMonth();
        int day1 = date1.getDate();
        int year2 = date2.getYear();
        int month2 = date2.getMonth();
        int day2 = date2.getDate();

        int diffYear = year2 - year1;
        int diffMonth = month2 - month1;
        int diffDay = day2 - day1;

        System.out.println("Year difference: " + diffYear);
        System.out.println("Month difference: " + diffMonth);
        System.out.println("Day difference: " + diffDay);
    }
}

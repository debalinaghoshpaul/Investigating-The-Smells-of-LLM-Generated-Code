```java
import java.util.HashMap;
import java.util.Map;

public class GetKeyValueOfMaxAndMinKey {

    public static void main(String[] args) {
        // create a map
        Map<String, Integer> map = new HashMap<>();

        // add key-value pairs to the map
        map.put("One", 1);
        map.put("Two", 2);
        map.put("Three", 3);
        map.put("Four", 4);
        map.put("Five", 5);

        // get the key-value mapping associated with the greatest key
        Map.Entry<String, Integer> maxEntry = map.entrySet().stream()
                .max((entry1, entry2) -> entry1.getKey().compareTo(entry2.getKey()))
                .get();

        // get the key-value mapping associated with the least key
        Map.Entry<String, Integer> minEntry = map.entrySet().stream()
                .min((entry1, entry2) -> entry1.getKey().compareTo(entry2.getKey()))
                .get();

        // print the key-value mappings
        System.out.println("Max Entry: " + maxEntry.getKey() + " -> " + maxEntry.getValue());
        System.out.println("Min Entry: " + minEntry.getKey() + " -> " + minEntry.getValue());
    }
}
```

        System.out.print("Enter the other two sides of the triangle: ");

        int b = scanner.nextInt();
        System.out.println("The area of the triangle is: " + calculateArea(s, a, b));
    }

    public static int calculateArea(int s, int a, int b) {
        int area = (s * (s - a - b)) / 2;
        return area;

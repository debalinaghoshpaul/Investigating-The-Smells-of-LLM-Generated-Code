```python
import sys

def main(filename):
    with open(filename) as f:
        text = f.read()
    words = text.split()
    word_counts = {}
    for word in words:
        word_counts[word] = word_counts.get(word, 0) + 1

    for word, count in word_counts.items():
        print(f'{word:20s} {count:10d}')

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print(f'Usage: {sys.argv[0]} <filename>')
        sys.exit()

    main(sys.argv[1])
```

```java
import java.util.ArrayList;

public class PrintArrayListElementsByPosition {

    public static void main(String[] args) {
        // create an ArrayList
        ArrayList<String> list = new ArrayList<>();

        // add elements to the ArrayList
        list.add("apple");
        list.add("banana");
        list.add("cherry");
        list.add("durian");
        list.add("elderberry");

        // print the elements of the ArrayList using the elements' position
        for (int i = 0; i < list.size(); i++) {
            System.out.println("Element at position " + i + ": " + list.get(i));
        }
    }
}
```

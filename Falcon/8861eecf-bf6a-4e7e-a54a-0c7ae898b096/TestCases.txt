```java
import java.util.Scanner;

public class Remove2ndCharacter {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Get the input string from the user
        System.out.println("Enter a string: ");
        String inputString = scanner.nextLine();

        // Find the substring of length three starting with 'z' and ending with 'g'
        int startIndex = inputString.indexOf("z");
        int endIndex = inputString.indexOf("g");
        if (startIndex == -1 || endIndex == -1 || endIndex - startIndex < 2) {
            System.out.println("The substring 'z...g' is not present in the input string.");
            return;
        }
        String substring = inputString.substring(startIndex, endIndex + 1);

        // Remove the 2nd character from the substring
        String newSubstring = substring.substring(0, 1) + substring.substring(2);

        // Replace the old substring with the new substring in the input string
        String newString = inputString.replace(substring, newSubstring);

        // Print the new string
        System.out.println("The new string is: " + newString);
    }
}
```

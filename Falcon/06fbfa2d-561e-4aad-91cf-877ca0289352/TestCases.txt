```java
import java.util.Scanner;

public class SubstringSurroundings {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        // Get the input string from the user
        System.out.println("Enter a non-empty string: ");
        String str = input.nextLine();

        // Get the substring from the user
        System.out.println("Enter a non-empty substring: ");
        String subStr = input.nextLine();

        // Check if the substring is contained in the string
        if (!str.contains(subStr)) {
            System.out.println("The substring is not contained in the string.");
            return;
        }

        // Create a new string to store the result
        StringBuilder result = new StringBuilder();

        // Iterate over the string and add each character before and after the substring to the result string
        for (int i = 0; i < str.length(); i++) {
            if (i < str.indexOf(subStr) || i > str.indexOf(subStr) + subStr.length() - 1) {
                result.append(str.charAt(i));
            }
        }

        // Print the result string
        System.out.println("The resulting string is: " + result.toString());
    }
}
```

Test cases:

* **Normal case:**
    * Input string: "Hello world"
    * Substring: "llo"
    * Output: "Heorld"
* **Corner cases:**
    * Input string: "Hello"
    * Substring: "llo"
    * Output: "Heo"
    * Input string: "Hello world"
    * Substring: "a"
    * Output: "Hello world"
    * Input string: "Hello world"
    * Substring: "Hello world"
    * Output: ""
